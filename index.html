<script type="module">

import {initializeApp} from "https://www.gstatic.com/firebasejs/12.17.1/firebase-app.js";

import {getAuth,onAuthStateChanged,signInWithEmailAndPassword,createUserWithEmailAndPassword,signOut} from "https://www.gstatic.com/firebasejs/12.17.1/firebase-auth.js";

import {initializeFirestore,persistentLocalCache,collection,doc,setDoc,deleteDoc,onSnapshot} from "https://www.gstatic.com/firebasejs/12.17.1/firebase-firestore.js";


// STEP: paste your own config from the Firebase console here

  authDomain: "my-money-10660.firebaseapp.com",

  projectId: "my-money-10660",

  storageBucket: "my-money-10660.firebasestorage.app",

  messagingSenderId: "842972782467",

  appId: "1:842972782467:web:9c4b5a7773780d190a60b6"

};

const app=initializeApp(firebaseConfig);

const auth=getAuth(app);

const db=initializeFirestore(app,{localCache:persistentLocalCache()}); // also works offline


const CATS={out:["Food","Transport","Bills","Shopping","Health","Fun","Other"],in:["Salary","Gift","Other income"]};

let txs=[],cur="$",type="out",view=new Date(),uid=null,unsub=null;view.setDate(1);

try{cur=localStorage.getItem("mm_cur")||"$"}catch(e){}

const $=id=>document.getElementById(id);

const money=n=>cur+n.toLocaleString(undefined,{minimumFractionDigits:2,maximumFractionDigits:2});

const iso=d=>d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0");

const esc=s=>s.replace(/[&<>"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));

const txCol=()=>collection(db,"users",uid,"transactions");


function render(){

  $("cur").value=cur;

  $("mlabel").textContent=view.toLocaleDateString(undefined,{month:"long",year:"numeric"});

  const key=iso(view).slice(0,7);

  const m=txs.filter(t=>t.date.startsWith(key)).sort((a,b)=>b.date.localeCompare(a.date)||b.id-a.id);

  const inc=m.filter(t=>t.type==="in").reduce((s,t)=>s+t.amount,0);

  const exp=m.filter(t=>t.type==="out").reduce((s,t)=>s+t.amount,0);

  $("bal").textContent=(inc-exp<0?"−":"")+money(Math.abs(inc-exp));

  $("inc").textContent=money(inc);$("exp").textContent=money(exp);

  const by={};m.filter(t=>t.type==="out").forEach(t=>by[t.cat]=(by[t.cat]||0)+t.amount);

  const rows=Object.entries(by).sort((a,b)=>b[1]-a[1]);

  $("cats").innerHTML=rows.length?rows.map(([c,v])=>`<div class="cat"><div class="l"><span>${esc(c)}</span><b>${money(v)}</b></div><div class="bar"><i style="width:${v/exp*100}%"></i></div></div>`).join(""):'<div class="empty">No spending yet this month.</div>';

  $("list").innerHTML=m.length?m.map(t=>`<div class="tx"><div class="m"><b>${esc(t.note||t.cat)}</b><span>${esc(t.cat)} · ${new Date(t.date+"T00:00").toLocaleDateString(undefined,{month:"short",day:"numeric"})}</span></div><div class="amt ${t.type}">${t.type==="in"?"+":"−"}${money(t.amount)}</div><button class="del" data-id="${t.id}" aria-label="Delete">×</button></div>`).join(""):'<div class="empty">Tap “+ Add” to record your first transaction.</div>';

}

function setType(t){type=t;$("tOut").classList.toggle("on",t==="out");$("tIn").classList.toggle("on",t==="in");$("cat").innerHTML=CATS[t].map(c=>`<option>${c}</option>`).join("")}


// ---- Sync: sign-in state and live data ----

onAuthStateChanged(auth,user=>{

  if(unsub){unsub();unsub=null}

  if(user){

    uid=user.uid;$("login").close();

    unsub=onSnapshot(txCol(),snap=>{txs=snap.docs.map(d=>d.data());render()});

  }else{

    uid=null;txs=[];render();$("pw").value="";$("login").showModal();

  }

});

$("login").addEventListener("cancel",e=>e.preventDefault());

const authErr=e=>{$("err").textContent=({"auth/invalid-credential":"Wrong email or password.","auth/email-already-in-use":"That email already has an account. Tap Sign in.","auth/weak-password":"Use a password with at least 6 characters.","auth/invalid-email":"Enter a valid email address."})[e.code]||"Could not sign in. Check your connection and try again."};

$("signin").onclick=()=>{$("err").textContent="";signInWithEmailAndPassword(auth,$("em").value.trim(),$("pw").value).catch(authErr)};

$("signup").onclick=()=>{$("err").textContent="";createUserWithEmailAndPassword(auth,$("em").value.trim(),$("pw").value).catch(authErr)};

$("signout").onclick=()=>signOut(auth);


// ---- Add and delete write to Firestore ----

$("add").onclick=()=>{setType("out");$("amount").value="";$("note").value="";$("date").value=iso(new Date());$("dlg").showModal();$("amount").focus()};

$("tOut").onclick=()=>setType("out");$("tIn").onclick=()=>setType("in");

$("cancel").onclick=()=>$("dlg").close();

$("save").onclick=()=>{

  const a=parseFloat($("amount").value);

  if(!(a>0)){$("amount").focus();return}

  const t={id:Date.now(),type,amount:a,cat:$("cat").value,note:$("note").value.trim(),date:$("date").value||iso(new Date())};

  setDoc(doc(db,"users",uid,"transactions",String(t.id)),t);

  $("dlg").close();

};

$("list").onclick=e=>{const b=e.target.closest(".del");if(!b)return;if(confirm("Delete this transaction?"))deleteDoc(doc(db,"users",uid,"transactions",b.dataset.id))};

$("prev").onclick=()=>{view.setMonth(view.getMonth()-1);render()};

$("next").onclick=()=>{view.setMonth(view.getMonth()+1);render()};

$("cur").onchange=e=>{cur=e.target.value;try{localStorage.setItem("mm_cur",cur)}catch(x){}render()};

setType("out");render();

</script>
